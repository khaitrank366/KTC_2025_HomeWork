YÊU CẦU 9: Đếm số user theo từng độ tuổi
------------------------------------------
db.users.aggregate([
  { $group: { _id: "$age", count: { $sum: 1 } } }
]);

------------------------------------------
YÊU CẦU 10: Tính tuổi trung bình của user >= 25
------------------------------------------
db.users.aggregate([
  { $match: { age: { $gte: 25 } } },
  { $group: { _id: null, averageAge: { $avg: "$age" } } }
]);
